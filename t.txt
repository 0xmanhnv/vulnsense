// Package domain contains the core business models and logic.
package domain

import "time"

// Asset is a polymorphic entity representing anything that can have vulnerabilities.
// It serves as a generic container, with specific data stored in the Details field.
type Asset struct {
	ID                string             `json:"id"`
	Name              string             `json:"name"`
	Type              AssetType          `json:"type"` // Crucial for determining the type of data in Details.
	Status            string             `json:"status,omitempty"`
	Environment       string             `json:"environment,omitempty"`
	OperatingSystem   OS                 `json:"operating_system,omitempty"`
	Cloud             *CloudInfo         `json:"cloud,omitempty"`
	OwnerTeam         string             `json:"owner_team,omitempty"`
	Location          string             `json:"location,omitempty"`
	FirstSeen         time.Time          `json:"first_seen,omitempty"`
	LastSeen          time.Time          `json:"last_seen,omitempty"`
	InstalledProducts []InstalledProduct `json:"installed_products,omitempty"`
	Tags              []string           `json:"tags,omitempty"`
	Metadata          map[string]any     `json:"metadata,omitempty"`

	// Details holds the specific information for the asset type (e.g., HostDetails, GitRepositoryDetails).
	// Using `any` makes this field polymorphic.
	Details any `json:"details"`
}

// OS represents the operating system information of an asset.
// This is a Value Object.
type OS struct {
	Name          string // e.g., "Ubuntu", "Windows Server"
	Version       string // e.g., "22.04", "2022"
	KernelVersion string // e.g., "5.15.0-101-generic"
	Architecture  string // e.g., "amd64", "arm64"
}

// CloudInfo represents cloud-specific details for an asset.
// This is a Value Object.
type CloudInfo struct {
	Provider   string // e.g., "AWS", "Azure", "GCP"
	Region     string // e.g., "us-east-1"
	InstanceID string // The provider-specific instance ID
	AccountID  string // The provider-specific account ID
}

// InstalledProduct represents a specific piece of software installed on an asset.
// This is a Value Object within the Asset aggregate.
type InstalledProduct struct {
	Name          string   // The name of the product, e.g., "openssl"
	Version       string   // The specific version, e.g., "1.1.1k"
	CPE           string   // Common Platform Enumeration, if available
	Vendor        string   // e.g., "OpenSSL Software Foundation"
	InstallPath   string   // Filesystem path, e.g., "/usr/bin/openssl"
	PackageSource string   // How it was installed, e.g., "deb", "rpm", "brew"
	Language      string   // If it's a library, e.g., "java", "python", "go"
	Signed        bool     // Whether the package is signed
	Tags          []string // Optional tags for categorization, e.g., ["web", "crypto"]
}

// HasProduct checks if a specific product is installed on the asset.
// This is a business logic method for the Asset entity.
func (a *Asset) HasProduct(productName string) bool {
	for _, p := range a.InstalledProducts {
		if p.Name == productName {
			return true
		}
	}
	return false
}
